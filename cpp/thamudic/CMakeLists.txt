cmake_minimum_required(VERSION 3.20)
project(ChimeraThamudic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(chimera_thamudic STATIC src/thamudic.cpp)
target_include_directories(chimera_thamudic PUBLIC include)
add_executable(thamudic_core_test tests/core_test.cpp)
target_link_libraries(thamudic_core_test PRIVATE chimera_thamudic)
enable_testing()
add_test(NAME thamudic_core_test COMMAND thamudic_core_test)
