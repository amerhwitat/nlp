cmake_minimum_required(VERSION 3.20)
project(AVRSGPU LANGUAGES CXX)
add_library(avrs_gpu STATIC src/gpu_backend.cpp)
target_include_directories(avrs_gpu PUBLIC include)
target_compile_features(avrs_gpu PUBLIC cxx_std_20)
option(AVRS_ENABLE_CUDA "Build CUDA kernels" OFF)
option(AVRS_ENABLE_OPENGL "Enable OpenGL shader assets" ON)
option(AVRS_ENABLE_DIRECTX12 "Enable DirectX 12 shader assets on Windows" ON)
if(AVRS_ENABLE_CUDA AND CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_library(avrs_cuda OBJECT cuda/historical_tensor.cu)
endif()
